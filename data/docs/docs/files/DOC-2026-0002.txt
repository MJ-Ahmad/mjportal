<script>
(() => {
  "use strict";

  const LAYOUTS = {
    "1441H": "Madina-1441H",
    "1405H": "Madina-1405H",
    "IndoPak": "IndoPak-15"
  };

  const $ = (id) => document.getElementById(id);

  const rightPage = $("page-right");
  const leftPage = $("page-left");

  const folioRight = $("folio-right");
  const folioLeft = $("folio-left");

  const slider = $("slider");
  const pageInput = $("pageno");

  const prevBtn = $("prev");
  const nextBtn = $("next");

  const where = $("where");
  const status = $("status");
  const layoutSelect = $("layout");

  let manifest = null;
  let currentPage = 1;
  let totalPages = 604;

  const params = new URLSearchParams(location.search);

  let currentLayout = params.get("layout") || "1441H";

  if (!LAYOUTS[currentLayout]) {
    currentLayout = "1441H";
  }

  layoutSelect.value = currentLayout;

  function savePage(page) {
    localStorage.setItem(
      `quran-reader-${currentLayout}`,
      String(page)
    );
  }

  function loadPagePosition() {
    const hashPage = parseInt(
      location.hash.replace("#", ""),
      10
    );

    if (!isNaN(hashPage)) {
      return hashPage;
    }

    const saved = parseInt(
      localStorage.getItem(
        `quran-reader-${currentLayout}`
      ),
      10
    );

    return saved || 1;
  }

  function loadLibrary() {
    return new Promise((resolve, reject) => {
      const existing = document.querySelector(
        'script[data-quran-lib]'
      );

      if (existing) {
        resolve();
        return;
      }

      const script = document.createElement("script");

      script.src = "./dist/quran-madina-html.js";
      script.dataset.quranLib = "1";

      script.setAttribute(
        "data-layout",
        currentLayout
      );

      script.setAttribute(
        "data-font-size",
        "20"
      );

      script.onload = resolve;
      script.onerror = reject;

      document.head.appendChild(script);
    });
  }

  async function loadManifest() {
    const path =
      "./assets/db/" +
      LAYOUTS[currentLayout] +
      "/manifest.json";

    const response = await fetch(path);

    if (!response.ok) {
      throw new Error(
        "Manifest not found: " + path
      );
    }

    manifest = await response.json();

    totalPages = manifest.n_pages || 604;

    slider.max = totalPages;
    pageInput.max = totalPages;
  }

  function arabicNumber(num) {
    return String(num).replace(/\d/g, d =>
      "٠١٢٣٤٥٦٧٨٩"[d]
    );
  }

  function spreadPages(page) {
    if (
      window.innerWidth <
      window.innerHeight
    ) {
      return [page];
    }

    return page % 2
      ? [page, page + 1]
      : [page - 1, page];
  }

  function updateControls() {
    slider.value = currentPage;
    pageInput.value = currentPage;

    prevBtn.disabled = currentPage <= 1;
    nextBtn.disabled =
      currentPage >= totalPages;

    location.hash = currentPage;

    savePage(currentPage);
  }

  function updateLocationInfo() {
    if (
      !manifest ||
      !manifest.pages
    ) {
      where.textContent =
        "Page " + currentPage;
      return;
    }

    const data =
      manifest.pages[currentPage - 1];

    if (!data) {
      where.textContent =
        "Page " + currentPage;
      return;
    }

    const sura =
      Number(data[0] || 0) + 1;

    where.textContent =
      "Sura " +
      sura +
      " · Page " +
      currentPage;
  }

  function showPage(page) {
    page = Math.max(
      1,
      Math.min(
        totalPages,
        parseInt(page, 10) || 1
      )
    );

    currentPage = page;

    const spread =
      spreadPages(page);

    rightPage.setAttribute(
      "page",
      spread[0]
    );

    folioRight.textContent =
      arabicNumber(spread[0]);

    if (
      spread.length > 1 &&
      spread[1] <= totalPages
    ) {
      leftPage.setAttribute(
        "page",
        spread[1]
      );

      folioLeft.textContent =
        arabicNumber(spread[1]);

      leftPage.parentElement.style.display =
        "";
    } else {
      leftPage.parentElement.style.display =
        "none";
    }

    updateControls();
    updateLocationInfo();

    document.title =
      "Page " +
      page +
      " · Quran Reader";

    status.textContent = "";
  }

  function nextPage() {
    showPage(currentPage + 1);
  }

  function prevPage() {
    showPage(currentPage - 1);
  }

  nextBtn.addEventListener(
    "click",
    nextPage
  );

  prevBtn.addEventListener(
    "click",
    prevPage
  );

  slider.addEventListener(
    "input",
    e => {
      pageInput.value =
        e.target.value;
    }
  );

  slider.addEventListener(
    "change",
    e => {
      showPage(e.target.value);
    }
  );

  pageInput.addEventListener(
    "change",
    e => {
      showPage(e.target.value);
    }
  );

  layoutSelect.addEventListener(
    "change",
    () => {
      const p =
        new URLSearchParams(
          location.search
        );

      p.set(
        "layout",
        layoutSelect.value
      );

      location.search =
        p.toString();
    }
  );

  document.addEventListener(
    "keydown",
    e => {
      if (
        e.key ===
          "ArrowLeft" ||
        e.key === "PageDown"
      ) {
        nextPage();
      }

      if (
        e.key ===
          "ArrowRight" ||
        e.key === "PageUp"
      ) {
        prevPage();
      }

      if (e.key === "Home") {
        showPage(1);
      }

      if (e.key === "End") {
        showPage(totalPages);
      }
    }
  );

  window.addEventListener(
    "hashchange",
    () => {
      const page =
        parseInt(
          location.hash.slice(1),
          10
        );

      if (
        !isNaN(page) &&
        page !== currentPage
      ) {
        showPage(page);
      }
    }
  );

  window.addEventListener(
    "resize",
    () => {
      showPage(currentPage);
    }
  );

  async function init() {
    try {
      status.textContent =
        "Loading Quran...";

      await loadLibrary();
      await loadManifest();

      showPage(
        loadPagePosition()
      );

      status.textContent = "";
    } catch (err) {
      console.error(err);

      status.textContent =
        "Failed to load Quran data.";
    }
  }

  init();
})();
</script>
